package org.openapitools.controllers;

import org.openapitools.api.ApiApi;
import org.openapitools.model.Etudiant;
import org.openapitools.model.LogementCrous;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ResponseStatusException;

import java.util.Map;

@RestController
public class ApiController implements ApiApi {

    private static final Map<String, Etudiant> ETUDIANTS = Map.of(
        "2300001", etudiant("1", "2300001", "Diop", "Maodo", "SATIC", "771234567",
                "maodo.diop@uadb.edu.sn", logement("B", "102", "A")),
        "2300002", etudiant("2", "2300002", "Ndiaye", "Aminata", "SEFS", "781234567",
                "aminata.ndiaye@uadb.edu.sn", logement("A", "215", "B")),
        "2200003", etudiant("3", "2200003", "Fall", "Cheikh", "SJP", "761234567",
                "cheikh.fall@uadb.edu.sn", null) // étudiant non logé
    );

    @Override
    public ResponseEntity<Etudiant> getEtudiantByMatricule(String matricule) {
        if (matricule == null || !matricule.matches("^[0-9]{7}$")) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST,
                    "Format de matricule invalide (doit comporter exactement 7 chiffres).");
        }
        Etudiant etudiant = ETUDIANTS.get(matricule);
        if (etudiant == null) {
            throw new ResponseStatusException(HttpStatus.NOT_FOUND,
                    "Aucun étudiant trouvé avec ce code permanent.");
        }
        return ResponseEntity.ok(etudiant);
    }

    private static LogementCrous logement(String pavillon, String numeroChambre, String lit) {
        LogementCrous l = new LogementCrous();
        l.setPavillon(pavillon);
        l.setNumeroChambre(numeroChambre);
        l.setLit(lit);
        return l;
    }

    private static Etudiant etudiant(String id, String matricule, String lastname, String firstname,
                                     String ufr, String telephone, String emailInstitutionnel,
                                     LogementCrous logementCrous) {
        Etudiant e = new Etudiant();
        e.setId(id);
        e.setMatricule(matricule);
        e.setLastname(lastname);
        e.setFirstname(firstname);
        e.setUfr(ufr);
        e.setTelephone(telephone);
        e.setEmailInstitutionnel(emailInstitutionnel);
        e.setLogementCrous(logementCrous);
        return e;
    }
}
=================== src/main/java/sn/uadb/controller/ApiExceptionHandler.java ===================
package org.openapitools.controllers;

import org.openapitools.model.ApiError;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.server.ResponseStatusException;

import java.time.OffsetDateTime;

@RestControllerAdvice
public class ApiExceptionHandler {

    @ExceptionHandler(ResponseStatusException.class)
    public ResponseEntity<ApiError> handle(ResponseStatusException ex) {
        ApiError err = new ApiError();
        err.setCode(ex.getStatusCode().value());
        err.setMessage(ex.getReason());
        err.setTimestamp(OffsetDateTime.now());
        return ResponseEntity.status(ex.getStatusCode()).body(err);
    }
}

package sn.uadb.filter;

import java.io.IOException;
import java.time.Instant;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicInteger;

import jakarta.servlet.Filter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;

public class RateLimitingFilter implements Filter {

    private static final Logger logger = LoggerFactory.getLogger(RateLimitingFilter.class);

    private static final int MAX_REQUESTS_PER_MINUTE = 2; // valeur de test
    private static final long WINDOW_MS = 60_000;
    private static final int CLEANUP_THRESHOLD = 10_000;

    private static final class Window {
        final long start;
        final AtomicInteger count = new AtomicInteger();
        Window(long start) { this.start = start; }
    }

    private final Map<String, Window> windows = new ConcurrentHashMap<>();

    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest httpRequest = (HttpServletRequest) request;
        HttpServletResponse httpResponse = (HttpServletResponse) response;

        String clientIp = httpRequest.getRemoteAddr();
        long now = System.currentTimeMillis();

        // Nouvelle fenêtre si aucune n'existe ou si l'ancienne a expiré (opération atomique)
        Window window = windows.compute(clientIp,
                (k, old) -> (old == null || now - old.start >= WINDOW_MS) ? new Window(now) : old);

        if (window.count.incrementAndGet() > MAX_REQUESTS_PER_MINUTE) {
            long retryAfter = Math.max(1, (window.start + WINDOW_MS - now + 999) / 1000);
            logger.warn("Rate limit exceeded for IP: {}", clientIp);

            httpResponse.setStatus(HttpStatus.TOO_MANY_REQUESTS.value());
            httpResponse.setHeader("Retry-After", String.valueOf(retryAfter));
            httpResponse.setContentType("application/json");
            httpResponse.setCharacterEncoding("UTF-8");
            httpResponse.getWriter().write(String.format(
                    "{\"error\":%d,\"message\":\"Rate limit exceeded. Please try again later.\",\"timestamp\":\"%s\"}",
                    HttpStatus.TOO_MANY_REQUESTS.value(), Instant.now()));
            return;
        }

        cleanupIfNeeded(now);
        chain.doFilter(request, response);
    }

    // Évite que la map grossisse indéfiniment
    private void cleanupIfNeeded(long now) {
        if (windows.size() > CLEANUP_THRESHOLD) {
            windows.values().removeIf(w -> now - w.start >= WINDOW_MS);
        }
    }
}

package org.openapitools ; 

import sn.uadb.filter.RateLimitingFilter;
import org.springframework.boot.web.servlet.FilterRegistrationBean; 
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.ComponentScan;
import org.springframework.context.annotation.Configuration;
import org.springframework.boot.SpringApplication;
import org.springframework.boot.autoconfigure.SpringBootApplication;


@SpringBootApplication(scanBasePackages = {"org.openapitools", "sn.uadb"})
public  class BackendApplication {

    public static void main(String[] args ){
       SpringApplication.run(BackendApplication.class,args) ;

    }

    @Bean
    public FilterRegistrationBean<RateLimitingFilter> rateLimitingFilter() {
        FilterRegistrationBean<RateLimitingFilter> registrationBean = new FilterRegistrationBean<>();
        registrationBean.setFilter(new RateLimitingFilter());
        registrationBean.addUrlPatterns("/api/*"); // Apply the filter to all URLs
        return registrationBean;
    }
}
